WhenBroadcastReceived(StartGame)
{
    Pen.Clear();
    floor = 1;
    floorlight = 0;
    turn = 0;
    List.DeleteAll(revealed tiles);
    List.DeleteAll(mapitems);
    List.DeleteAll(flooritemmount);
    Playerloc = 1180;
    Control.Wait(0.1);
    Call Generate items;
    Forever
    {
        Call ItemCheck("n");
        Pen.Clear();
        Call PaintMap;
        If (Operator.LetterOf(maps[floor], Playerloc) == 3)
        {
            Call Illu1;
        }
        Else
        {
            Call Illuminate;
        }
        prev pos = Playerloc;
        If (inventoryopen == 0)
        {
            Call Movement;
        }
        Else
        {
            Repeat Until (inventoryopen == 0)
            {
                Pen.Clear();
                Call PaintMap;
                Event.Broadcast("clearclones");
                Event.Broadcast("Paint Bottom");
            }
            Control.Wait(0);
        }
        If ((Operator.LetterOf(maps[floor], Playerloc) == 1) Or ((floor == 26) And (Not (List.ContainsItem(monstermaploc, "Q")))))
        {
            If List.ContainsItem(MH floors, floor)
            {
                Sound.StopAllSounds();
            }
            floor += 1;
            List.DeleteAll(revealed tiles);
            Playerloc = 1180;
            PlayerPowerUps = 1;
            player's status = 0;
            Event.Broadcast("Advance");
            Control.Wait(0.2);
            Sound.Play(Footsteps);
            floorlight = 0;
            If ((floor == 26) Or (floor == 40))
            {
                List.DeleteAll(revealed tiles);
                List.DeleteAll(mapitems);
                List.DeleteAll(flooritemmount);
                Call RevealMap;
            }
            Else
            {
                List.DeleteAll(mapitems);
                List.DeleteAll(flooritemmount);
                Call Generate items;
            }
            If List.ContainsItem(MH floors, floor)
            {
                Call RevealMap;
                List.Add(textlog, "It's a Monster House!");
            }
            Control.Wait(0.2);
        }
        If ((floor == 40) And (Not (List.ContainsItem(monstermaploc, "G"))))
        {
            Control.Wait(1);
            List.Add(textlog, "Congratulations! You have defeated the all-mighty bringer of evil *S U D O  G O D*!");
            List.Add(textlog, Operator.Join("Press R if you want to play yet again! Score: ", ((PlayerScorept1 + PlayerExp) + ((PlayerLevel * 1000) + 1493800))));
            Event.Broadcast("Ending");
            Forever
            {
                Event.Broadcast("clearclones");
                Event.Broadcast("Paint Bottom");
                Pen.Clear();
                Call PaintMap;
            }
        }
    }
}

                    Playerloc = prev pos;
                    moved = 0;
                }
            }
        }
        Else

        }
    }
    If (moved == 1)

        }
        If (player's status == 200)
        {
            player's status = 0;
            List.Add(textlog, "You regained conciousness.");
        }
    }
    moved = 0;
}

Define PastTurn (warp=false)
{
    turn += 1;
    Event.Broadcast("EnemyMovement");
    Control.Wait(0);
    Event.Broadcast("clearclones");
    Event.Broadcast("Paint Bottom");
    Pen.Clear();
    Call PaintMap;
    Wait Until ((Sensing.Timer() > (0.15 * (((monsternearby == 1) And Sensing.KeyPressed(x)) + 1))) Or (Sensing.KeyPressed(x) And (Not (Sensing.KeyPressed(z)))));
    Sensing.ResetTimer();
    If ((turn % (((6 / (((inventory[List.IndexOf(equipped, "Armband")] == 20) * 2) + 1)) / ((inventory[List.IndexOf(equipped, "Shield")] == 12) + 1)) * ((inventory[List.IndexOf(equipped, "Shield")] == 13) + 1))) == 0)
    {
        If (PlayerFull > 0)
        {
            PlayerFull += -1;
            If (PlayerFull == 20)
            {
                List.Add(textlog, "You're getting hungry...");
                Sound.Play(Growl);
            }
            If (PlayerFull == 10)
            {
                List.Add(textlog, "You're so hungry, you're feeling dizzy.");
                Sound.Play(Growl);
            }
            If (PlayerFull == 0)
            {
                List.Add(textlog, "Oh no! You're wasting away!");
                Sound.Play(Growl);
            }
        }
    }
    Control.Wait(0);
    If (PlayerHp < 1)
    {
        List.Add(textlog, "You died.");
        PlayerHp = 0;
        Repeat (30)
        {
            Event.Broadcast("clearclones");
            Event.Broadcast("Paint Bottom");
            Pen.Clear();
            Call PaintMap;
        }
        If List.ContainsItem(inventory, 30)
        {
            List.DeleteItem(inventorycount, List.IndexOf(inventory, 30));
            Sound.Play(Revive);
            List.DeleteItem(equipped, List.IndexOf(inventory, 30));
            List.DeleteItem(inventory, List.IndexOf(inventory, 30));
            PlayerHp = PlayerMaxHp;
            PlayerFull = 100;
            List.Add(textlog, "Revival Herb brought you back!");
            List.Add(textlog, "Revival Herb disappeared.");
            player's status = 0;
        }
        Else
        {
            Sound.Play(Game Over);
            List.Add(textlog, Operator.Join("Press R to play again. Score: ", ((PlayerScorept1 + PlayerExp) + ((PlayerLevel * 1000) + (floor * 12345)))));
            Repeat (99)
            {
                Event.Broadcast("clearclones");
                Event.Broadcast("Paint Bottom");
                Pen.Clear();
                Call PaintMap;
            }
            Wait Until (1 == 0);
        }
    }
    If (PlayerScorept1 > 1)
    {
        PlayerScorept1 += -1;
    }
    If (PlayerFull < 1)
    {
        List.Add(textlog, foodquotes[((turn % 2) + 1)]);
        PlayerHp += -1;
    }
    Else
    {
        If (inventory[List.IndexOf(equipped, "Armband")] == 20)
        {
            PlayerHp += 6;
        }
        Else
        {
            PlayerHp += (floor((turn * (PlayerMaxHp / 120))) - floor(((turn - 1) * (PlayerMaxHp / 120))));
        }
    }
    If (PlayerHp > PlayerMaxHp)
    {
        PlayerHp = PlayerMaxHp;
    }
    If ((turn % (40 - Operator.Round((floor / 2)))) == 1)
    {
        Event.Broadcast("Monster");
    }
    If (inventory[List.IndexOf(equipped, "Armband")] == 22)
    {
        PlayerExp += 1;
    }
}
