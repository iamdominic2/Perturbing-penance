//----- Broadcast received events -------------------------------------------------------------------------------------

WhenBroadcastReceived(clearclones)
{
    Control.DeleteThisClone();
}

WhenBroadcastReceived(GF)
{
    generatingmap = 0;
    Event.BroadcastAndWait("Generate");
    Event.Broadcast("StartGame");
    List.Add(textlog, "Use the arrow keys to move. Press Z to attack! Arrows + X to dash.");
    List.Add(textlog, "C or I for inventory, X to exit inventory, A to hide monsters, S for diagonal movement.");
    List.Add(textlog, "D to change direction without moving, Q to show minimap only. Mouse wheel or P/L to scroll messages.");
    List.Add(textlog, "M to hide/show minimap. Z + X to heal in place. If you want to view this help message again, press H!");
    Forever
    {
        If Sensing.KeyPressed(h)
        {
            List.Add(textlog, "Use the arrow keys to move. Press Z to attack! Arrows + X to dash, C or I for inventory");
            List.Add(textlog, "X to exit inventory, A to hide monsters, S for diagonal movement.");
            List.Add(textlog, "D to change direction without moving, Q to show minimap only.");
            List.Add(textlog, "M to hide/show minimap. Z+X to heal in place. If you want to view this again, press H!");
            Wait Until (Not (Sensing.KeyPressed(h)));
        }
    }
}

WhenBroadcastReceived(Paint Bottom)
{
    Costume.GoToFrontBack(front);
    Looks.SetEffectTo(BRIGHTNESS, 100);
    Call Write(Operator.Join(floor, "F")) at(-230);
    Call Write(Operator.Join("^ ", PlayerLevel)) at(-195);
    If (PlayerHp > PlayerMaxHp)
    {
        PlayerHp = PlayerMaxHp;
    }
    If (PlayerHp < 0)
    {
        PlayerHp = 0;
    }
    Call Write(Operator.Join("HP ", Operator.Join(Operator.Join(PlayerHp, "/"), PlayerMaxHp))) at(-156);
    Call Write(Operator.Join("&", PlayerExp)) at(-75);
    Call Write(Operator.Join("#  ", Operator.Join(Operator.Join(PlayerFull, "/"), PlayerMaxFull))) at(170);
    Call Write(Operator.Join(""   ", (PlayerPower + ((inventory[List.IndexOf(equipped, "Armband")] == 18) * 3)))) at(90);
    Call Write(Operator.Join("'  ", ((exp table[PlayerLevel] - PlayerExp) + 0))) at(10);
    If (PlayerHp > 0)
    {
        Pen.SetPenColorToColor(#ff0000);
        Pen.SetPenSizeTo(8);
        Pen.SetPenTo(color, ((PlayerHp / PlayerMaxHp) * 32));
        Motion.GoToXY(-160, -179);
        Pen.Down();
        Motion.ChangeXBy(((PlayerHp / PlayerMaxHp) * 63));
        Pen.Clear();
    }
}

WhenBroadcastReceived(sosis)
{
    Stop(other scripts in sprite);
    Control.DeleteThisClone();
}


//----- Custom blocks -------------------------------------------------------------------------------------------------

Define Write(string number or text) at(string x) (warp=true)
{
    Looks.SetSizeTo(40);
    Looks.Show();
    Motion.GoToXY(x, -173);
    counter1 = 0;
    Repeat (Variable.Length(number or text))
    {
        counter1 += 1;
        Looks.SwitchCostumeTo(Operator.LetterOf(number or text, counter1));
        Control.CreateCloneOf(_myself_);
        Motion.ChangeXBy((8 / ((Costume.GetNumberName(number) == 20) + 1.3)));
    }
    Looks.Hide();
}
